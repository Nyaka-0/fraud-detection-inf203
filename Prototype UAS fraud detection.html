"""

from datetime import datetime, time

BATAS_NOMINAL = 50_000_000       # transaksi di atas ini dianggap besar
JAM_MULAI_WAJAR = time(6, 0)     # jam wajar transaksi: 06:00
JAM_SELESAI_WAJAR = time(22, 0)  # sampai 22:00
BATAS_FREKUENSI = 3              # lebih dari 3x transaksi/hari dianggap mencurigakan

def deteksi_fraud(transaksi, riwayat_lokasi=None, jumlah_transaksi_hari_ini=0):
    alasan = []

    # Aturan 1: nominal besar
    if transaksi["jumlah"] > BATAS_NOMINAL:
        alasan.append(f"Nominal melebihi batas wajar (> Rp{BATAS_NOMINAL:,})")

    # Aturan 2: waktu transaksi tidak wajar
    jam_transaksi = transaksi["waktu"].time()
    if not (JAM_MULAI_WAJAR <= jam_transaksi <= JAM_SELESAI_WAJAR):
        alasan.append("Waktu transaksi di luar jam wajar (di luar 06:00-22:00)")

    # Aturan 3: lokasi berbeda drastis dari riwayat
    if riwayat_lokasi and transaksi["lokasi"] != riwayat_lokasi:
        alasan.append(f"Lokasi berbeda dari riwayat ({riwayat_lokasi} -> {transaksi['lokasi']})")

    # Aturan 4: frekuensi transaksi tinggi
    if jumlah_transaksi_hari_ini > BATAS_FREKUENSI:
        alasan.append(f"Frekuensi transaksi tinggi ({jumlah_transaksi_hari_ini}x hari ini)")

    status = "Mencurigakan (Fraud Alert)" if alasan else "Normal"

    return {
        "id": transaksi["id"],
        "status": status,
        "alasan": alasan if alasan else ["Tidak ada indikasi mencurigakan"]
    }
